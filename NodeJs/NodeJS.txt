    //servers.js
    import http from 'http';
    const server =  http.createServer((req, res) => {
        // res.write("Hello World, This is first day NodeJs Course");
        // res.write(req.method);
        // res.end();
        if(req.method === "GET") {
            if(req.url === "/") {
                res.writeHead(200, {"Content-Type": "text/html"});
                res.write("Hello World, This is first day NodeJs Course");
                res.end();
            }else if(req.url === "/about") {
                res.writeHead(200, {"Content-Type": "text/html"});
                res.write("This is about page");
                res.end();
            }else if(req.url === "/contact") {
                res.writeHead(200, {"Content-Type": "text/html"});
                res.write("This is contact page");
                res.end();
            }else {
                res.writeHead(404, {"Content-Type": "text/html"});
                res.write("Page not found");
                res.end();
            }
        }else if(req.method === "POST") {
            let body = "";
            if(req.url === "/login") {
                req.on("data", (data) => {
                    body+=data.toString();
                });
                req.on('end', () => {
                    console.log(body);
                    res.end(JSON.stringify({stats:"success", message: "Data received"}));
                });
            }
        }
    })

    server.listen(8080, () => {
        console.log("Server is running on port 8080");
    });

        Real-Time Use Case

        E-Commerce Order Management System
        Technologies Used
            Frontend: React
            Backend: Express.js
            Database: MySQL
            Authentication: JSON Web Token
            Session Management: Express Session
        Scenario:
        A company builds an online shopping platform where customers log in, browse products, add items to a cart, and place orders.
        The system must handle:
            Secure authentication
            User sessions
            API authorization
            Database storage
            Frontend interaction
        System Architecture
        React (Frontend)|
            |  API Requests
            vExpressJS Backend|
            | Session + JWT Verification
            vMySQL Database

        Step 1 — User Login
        Real-Time Flow
            User enters email and password in React login page.
            React sends login request to Express API.
                POST /api/login
            Express checks credentials in MySQL users table.
        Example Table
        id	email	password	role
        1	john@gmail.com	hashed_pwd	customer

            If valid:
                Express generates JWT Token
                Express creates Session
                Token sent to frontend

        Step 2 — JWT Token Generation
        Purpose: API Security
        When login succeeds:
        JWT Token created with:{
        userId: 101,
        role: "customer",
        loginTime: timestamp}
        Token is returned to React.
        React stores token in:
        localStorage

        Step 3 — Session Creation
        At the same time, Express creates a server session.
        Session stores:
        session = {
        userId: 101,
        cartItems: [],
        loginTime: "10:30 AM"}
        Why session is used:
        Maintain shopping cart
        Track logged-in user
        Store temporary data

        Step 4 — Product Browsing
        User opens Products Page in React.
        React sends API request:
        GET /api/productsAuthorization: Bearer <JWT_TOKEN> 
        Express performs:
        JWT Verification
        Fetch products from MySQL
        MySQL Table
        product_id	name	price	stock
        1	Laptop	70000	10
        2	Phone	20000	15
        Data returned to React and displayed.

        Step 5 — Add to Cart (Session Use Case)
        User clicks Add to Cart.
        React request:
        POST /api/cart/add
        Express does:
        req.session.cart.push(product)
        Session Example
        session = {
        userId: 101,
        cartItems: [
        {productId:1, qty:1},
        {productId:2, qty:2}
        ]
        }
        Session avoids storing temporary cart data in DB until checkout.

        Step 6 — Checkout
        User clicks Checkout.
        React sends:
        POST /api/order Authorization: Bearer token
        Express:
        Verifies JWT
        Reads cart from session
        Saves order into MySQL
        Orders Table
        order_id	user_id	total_price
        1001	101	110000
        Order Items Table
        order_id	product_id	quantity
        1001	1	1
        1001	2	2
        Step 7 — Logout
        User logs out.
        React calls:
        POST /api/logout
        Express:
        req.session.destroy()
        Frontend removes JWT token.
        User must login again.
